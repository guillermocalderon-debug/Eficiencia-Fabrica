        function parseRobustFloat(val) {
            if (val === undefined || val === null || val === "") return 0;
            let str = String(val).replace(/%/g, '').trim(); 
            if (str.includes(',') && !str.includes('.')) { str = str.replace(',', '.'); } 
            else if (str.includes('.') && str.includes(',')) { str = str.replace(/\./g, '').replace(',', '.'); }
            return parseFloat(str) || 0;
        }

        function normalizeDateToNum(dateInput) {
            if (!dateInput) return null;
            let str = String(dateInput).trim();
            if (str === "") return null;
            if (str.includes('T')) str = str.split('T')[0];
            if (str.includes('-')) {
                let p = str.split('-');
                if (p.length >= 3) {
                    if (p[0].length === 4) return parseInt(p[0] + p[1].padStart(2,'0') + p[2].substring(0,2).padStart(2,'0'), 10);
                    if (p[2].length === 4) return parseInt(p[2] + p[1].padStart(2,'0') + p[0].substring(0,2).padStart(2,'0'), 10);
                }
            }
            if (str.includes('/')) {
                let p = str.split(' ')[0].split('/');
                if (p.length >= 2) {
                    let d = p[0].padStart(2,'0'), m = p[1].padStart(2,'0'), y = p.length === 3 ? p[2] : String(new Date().getFullYear());
                    if (y.length === 2) y = "20" + y;
                    return parseInt(y + m + d, 10);
                }
            }
            return null;
        }

        function parseRosterData(rosterData) {
            if (!rosterData || rosterData.length === 0) return;
            let rosterMap = {};
            let dateCols = {};

            let isObject = !Array.isArray(rosterData[0]);
            if (isObject) {
                Object.keys(rosterData[0]).forEach(k => {
                    let d = normalizeDateToNum(k);
                    if(d) dateCols[k] = d;
                });
            } else {
                let bestRow = -1, maxDates = 0;
                for(let r=0; r < Math.min(10, rosterData.length); r++) {
                    let count = 0;
                    rosterData[r].forEach(c => { if(normalizeDateToNum(c)) count++; });
                    if(count > maxDates) { maxDates = count; bestRow = r; }
                }
                if(bestRow > -1) {
                    rosterData[bestRow].forEach((c, idx) => {
                        let d = normalizeDateToNum(c);
                        if(d) dateCols[idx] = d;
                    });
                }
            }

            for (let i = isObject ? 0 : 1; i < rosterData.length; i++) {
                let row = rosterData[i];
                let vals = isObject ? Object.values(row) : row;
                
                let opObj = null;
                for (let v of vals) {
                    let opLower = String(v).toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "");
                    for (let p of PERSONNEL) {
                        for (let k of p.keys) {
                            if (opLower.includes(k)) { opObj = p; break; }
                        }
                    }
                    if (opObj) break;
                }
                if (!opObj) continue;

                for (let key in dateCols) {
                    let dateNum = dateCols[key];
                    let asignacion = String(row[key] || "").trim().toUpperCase();
                    if (asignacion.includes("(") && (asignacion.includes("M)") || asignacion.includes("T)") || asignacion.includes("N)"))) {
                        let parts = asignacion.split("(");
                        let maqStr = parts[0].replace(/M/g, '').replace(/MAQ/g,'').trim();
                        let turnoLetra = parts[1].replace(')','').trim();
                        
                        if (!rosterMap[dateNum]) rosterMap[dateNum] = {};
                        maqStr.split('-').forEach(num => {
                            let n = parseInt(num, 10);
                            if (!isNaN(n)) {
                                let maqLabelFull = `Inyectora ${n}`;
                                if (!rosterMap[dateNum][maqLabelFull]) rosterMap[dateNum][maqLabelFull] = {};
                                rosterMap[dateNum][maqLabelFull][turnoLetra] = opObj;
                            }
                        });
                    }
                }
            }
            cachedRoster = rosterMap;
            localStorage.setItem(ROSTER_CACHE_KEY, JSON.stringify(rosterMap));
        }

        function descargarRosterJSONP() {
            return new Promise((resolve) => {
                const callbackName = 'rosterCallback_' + Math.round(100000 * Math.random());
                window[callbackName] = function(data) {
                    parseRosterData(data);
                    delete window[callbackName];
                    const scriptEl = document.getElementById(callbackName);
                    if(scriptEl) scriptEl.remove();
                    resolve();
                };

                const script = document.createElement('script');
                script.id = callbackName;
                script.src = `${ROSTER_API}?callback=${callbackName}&_t=${Date.now()}`;
                script.onerror = () => resolve(); 
                document.body.appendChild(script);
            });
        }

        function descargarDatosJSONP() {
            const callbackName = 'jsonpCallback_' + Math.round(100000 * Math.random());
            window[callbackName] = function(data) {
                localStorage.setItem(CACHE_KEY, JSON.stringify(data));
                renderizarDatos(data, false);
                delete window[callbackName];
                const scriptEl = document.getElementById(callbackName);
                if(scriptEl) scriptEl.remove();
            };

            const script = document.createElement('script');
            script.id = callbackName;
            script.src = `${API_URL}?callback=${callbackName}&_t=${Date.now()}`;
            document.body.appendChild(script);
        }

        function initDashboard() {
            const cachedData = localStorage.getItem(CACHE_KEY);
            const rCache = localStorage.getItem(ROSTER_CACHE_KEY);
            if (rCache) cachedRoster = JSON.parse(rCache);
            if (cachedData) renderizarDatos(JSON.parse(cachedData), true);

            descargarRosterJSONP().then(() => {
                descargarDatosJSONP();
            });

            setInterval(() => {
                descargarRosterJSONP().then(descargarDatosJSONP);
            }, 300000); 

            changeTimeframe('dia');
        }
